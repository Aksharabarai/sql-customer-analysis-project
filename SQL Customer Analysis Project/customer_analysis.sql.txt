#Create Database
CREATE DATABASE sales_analysis;
USE sales_analysis;

#Create Tables
#Customers Table
CREATE TABLE customers (
customer_id INT PRIMARY KEY,
customer_name VARCHAR(50),
city VARCHAR(50),
gender VARCHAR(10),
age INT
);

#Products Table
CREATE TABLE products (
product_id INT PRIMARY KEY,
product_name VARCHAR(50),
category VARCHAR(50),
price INT
);

#Orders Table
CREATE TABLE orders (
order_id INT PRIMARY KEY,
customer_id INT,
product_id INT,
quantity INT,
order_date DATE,
FOREIGN KEY (customer_id) REFERENCES customers(customer_id),
FOREIGN KEY (product_id) REFERENCES products(product_id)
);

#Important JOIN Queries
#Full Order Details
SELECT o.order_id,c.customer_name,p.product_name,o.quantity,o.order_date
FROM orders o
JOIN customers c ON o.customer_id=c.customer_id
JOIN products p ON o.product_id=p.product_id;

#Total Revenue
SELECT SUM(p.price * o.quantity) AS total_revenue
FROM orders o
JOIN products p ON o.product_id=p.product_id;

#Top Customers by Spending
SELECT c.customer_name,
SUM(p.price * o.quantity) AS spending
FROM orders o
JOIN customers c ON o.customer_id=c.customer_id
JOIN products p ON o.product_id=p.product_id
GROUP BY c.customer_name
ORDER BY spending DESC;

#Best Selling Products
SELECT p.product_name,
SUM(o.quantity) AS units_sold
FROM orders o
JOIN products p ON o.product_id=p.product_id
GROUP BY p.product_name
ORDER BY units_sold DESC;

#City-wise Revenue
SELECT c.city,
SUM(p.price * o.quantity) AS revenue
FROM orders o
JOIN customers c ON o.customer_id=c.customer_id
JOIN products p ON o.product_id=p.product_id
GROUP BY c.city;

#Monthly Sales
SELECT MONTH(order_date) AS month,
SUM(p.price * o.quantity) AS revenue
FROM orders o
JOIN products p ON o.product_id=p.product_id
GROUP BY MONTH(order_date);

#Rank Customers
SELECT c.customer_name,
SUM(p.price * o.quantity) spending,
RANK() OVER(ORDER BY SUM(p.price * o.quantity) DESC) ranking
FROM orders o
JOIN customers c ON o.customer_id=c.customer_id
JOIN products p ON o.product_id=p.product_id
GROUP BY c.customer_name;

#Repeat Customers
SELECT c.customer_name,
COUNT(o.order_id) total_orders
FROM orders o
JOIN customers c ON o.customer_id=c.customer_id
GROUP BY c.customer_name
HAVING COUNT(o.order_id) > 1;

#Second Highest Spending Customer
SELECT *
FROM (
SELECT c.customer_name,
SUM(p.price * o.quantity) spending,
DENSE_RANK() OVER(ORDER BY SUM(p.price * o.quantity) DESC) rnk
FROM orders o
JOIN customers c ON o.customer_id=c.customer_id
JOIN products p ON o.product_id=p.product_id
GROUP BY c.customer_name
) x
WHERE rnk=2;
